# Experimental drivers retain their existing CLI and numerical compile definitions.
foreach(driver run_ltv_gv_evaluation run_ltv_gv_production)
    add_executable(${driver} run_ltv_feature_passive.cpp)
    target_compile_definitions(${driver} PRIVATE LTV_GV_EVALUATION=1)
    if(driver STREQUAL "run_ltv_gv_production")
        target_compile_definitions(${driver} PRIVATE LTV_GV_PRODUCTION_LEVEL=1)
    endif()
    ament_target_dependencies(${driver} ${ament_libraries})
    target_link_libraries(${driver} ov_msckf_lib ${thirdparty_libraries})
    install(TARGETS ${driver} DESTINATION lib/${PROJECT_NAME})
endforeach()
